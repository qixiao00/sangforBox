obj.namestring--接口名称
最大长度:48
obj.descriptionstring--接口描述
最大长度:96
obj.ipv4object--接口ipv4相关配置
obj.ipv4.ipv4Modestring--IPv4地址获取类型
枚举值:[
STATIC:静态配置
DHCP:dhcp方式获取
PPPOE:pppoe方式获取
]
obj.ipv4.staticIpobject[]--ipv4地址列表
obj.ipv4.staticIp.ipaddressobject--IPv4地址
格式:ip-range
obj.ipv4.staticIp.ipaddress.startstring是ip范围的起始位置
格式:ip-address
obj.ipv4.staticIp.ipaddress.endstring否ip范围的终止位置
格式:ip-address
obj.ipv4.staticIp.ipaddress.bitsuint8否ip掩码简写
obj.ipv4.staticIp.isSyncbool--是否不同步到备机的选项
obj.ipv4.dhcpobject--dhcp4 client配置
obj.ipv4.dhcp.dhcp4IpListobject[]--dhcp4的ipv4地址列表
obj.ipv4.dhcp.dhcp4IpList.ipaddressobject--IPv4地址
格式:ip-range
obj.ipv4.dhcp.dhcp4IpList.ipaddress.startstring是ip范围的起始位置
格式:ip-address
obj.ipv4.dhcp.dhcp4IpList.ipaddress.endstring否ip范围的终止位置
格式:ip-address
obj.ipv4.dhcp.dhcp4IpList.ipaddress.bitsuint8否ip掩码简写
obj.ipv4.dhcp.dhcp4IpList.isSyncbool--是否不同步到备机的选项
obj.ipv4.dhcp.gatewaybool--是否设置dhcp gateway
obj.ipv4.dhcp.dnsbool--是否设置dhcp dns
obj.ipv4.dhcp.unicastbool--是否设置dhcp单播
obj.ipv4.pppoeobject--pppoe配置
obj.ipv4.pppoe.pppoeGroupNamestring--pppoe group name
obj.ipv4.pppoe.pppoeIpListobject[]--pppoe的ipv4地址列表
obj.ipv4.pppoe.pppoeIpList.ipaddressobject--IPv4地址
格式:ip-range
