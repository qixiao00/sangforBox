objobject是ssl代理策略配置
obj.namestring--策略名
最小长度:1
最大长度:95
obj.enablebool--是否启用
默认值:true
obj.zonesstring[]--区域
obj.ipgroupsstring[]--网络对象
obj.sessionReusebool--是否启用会话复用
默认值:true
obj.sessionCacheNumuint32--最大会话数
默认值:2000
obj.sessionTimeoutuint32--会话超时时间
默认值:1800
obj.policyTypestring--ssl代理策略类型
默认值:server
枚举值:[
server:服务器场景
internet:上网场景
user auth:认证模块使用
_
urlfile filter:URL过滤模块使用
_
]
obj.certsstring[]--服务器证书
obj.internetobject--上网场景配置
obj.internet.websiteAllbool--是否全部站点
默认值:true
obj.internet.websiteCategorystring[]--站点的类别
obj.internet.websiteCategoryIduint8[]--url类型id数组
obj.internet.installCertbool--是否提示用户安装证书
obj.internet.certUrlstring--证书地址
最小长度:0
最大长度:4096
obj.serverobject--服务器场景配置
obj.server.serversobject[]--服务器IP/端口
obj.server.servers.ipstring--ipv4或者ipv6地址
格式:ip-address
